<script>
(async () => {
  const $ = (id) => document.getElementById(id);
  const say = (t, bad) => { $('msg').textContent = t; $('msg').className = 'msg' + (bad ? ' bad' : ''); };
  const esc = (s) => String(s == null ? '' : s).replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' })[c]);
  const COLS = ['todo', 'doing', 'done'];
  const [db, user] = await Promise.all([claude.use('db'), claude.use('user')]);
  if (!db) { say('No database in this view.', true); return; }
  const me = user ? await user.me() : null;
  const uid = user ? await user.id() : null;
  $('who').textContent = me ? `${me.name || me.id} · ${me.level}` : '';
  const canWrite = me ? ['interact', 'admin', 'owner'].includes(me.level) : false;
  if (!canWrite) { $('add').hidden = true; }
  if (user && await user.canEdit()) $('rename').hidden = false;

  db.doc('settings/board').onSnapshot((s) => { $('name').textContent = (s.data() || {}).name || 'Board'; });
  $('rename').onclick = async () => {
    const name = window.__lab_prompt ? await window.__lab_prompt('Board name', $('name').textContent) : null;
    if (name) db.doc('settings/board').set({ name }).catch((e) => say(e.message, true));
  };

  const names = {};
  db.collection('cards').orderBy('at').onSnapshot(async (snap) => {
    const ids = [...new Set(snap.docs.map((d) => d.data().by).filter((x) => x && !(x in names)))];
    if (ids.length && user) Object.entries(await user.profiles(ids)).forEach(([id, p]) => { names[id] = p.name || 'Someone'; });
    for (const c of COLS) $(c).innerHTML = '';
    for (const d of snap.docs) {
      const c = d.data(); const i = COLS.indexOf(c.col);
      const el = document.createElement('div'); el.className = 'card';
      el.innerHTML = `<div class="t">${esc(c.text)}</div><div class="by">${esc(names[c.by] || '')}</div>` + (canWrite ? `<div class="row"><button data-m="-1" ${i <= 0 ? 'disabled' : ''}>←</button><button data-m="1" ${i >= 2 ? 'disabled' : ''}>→</button><button data-x>Remove</button></div>` : '');
      el.onclick = async (ev) => {
        const b = ev.target.closest('button'); if (!b) return;
        try {
          if (b.dataset.x !== undefined) await db.doc('cards/' + d.id).delete();
          else await db.doc('cards/' + d.id).update({ col: COLS[i + Number(b.dataset.m)] });
        } catch (e) { say(`${e.code}: ${e.message}`, true); }
      };
      $(COLS[Math.max(0, i)]).appendChild(el);
    }
  }, (e) => say(e.message, true));

  $('add').onsubmit = async (ev) => {
    ev.preventDefault();
    const text = $('text').value.trim(); if (!text) return;
    try { await db.collection('cards').add({ text, col: 'todo', by: uid, at: Date.now() }); $('text').value = ''; }
    catch (e) { say(`${e.code}: ${e.message}`, true); }
  };

  if (uid) {
    const notes = db.doc(`data/users/${uid}/notes`);
    const s = await notes.get(); $('notes').value = s.exists ? s.data().text : '';
    let timer = null;
    $('notes').oninput = () => { clearTimeout(timer); timer = setTimeout(() => notes.set({ text: $('notes').value }).then(() => say('Notes saved.')).catch((e) => say(`${e.code}: ${e.message}`, true)), 600); };
  } else $('notes').disabled = true;
})();
</script>
