ui.when(c => c.state.open, ui.text('…'))     // RUNTIME predicate

.title('…') .caption(…) .footer(…) .hint('…')    // window and chrome
.on('click', (ev, c) => …)                       // runtime; c is the context: state param db board window set
.bind(c => c.state.n)                            // runtime, reactive; the node's deps are what the lambda read
.map(b => b.pad(4))                              // build-time; transform this builder and return it
.use(fragment)                                   // mix a fragment in: new methods, attributes, decorations
.id('results') .ref('results')                   // name a node; point at one (a second parent → DAG, not a tree)
.draggable({ kind }) .at(x, y) .near(window)     // placement on the board

// Three ways out
ui.stack(…).build()        // → IR: plain objects, inspectable, diffable
ui.stack(…).mount(el)      // → a live view with .state and .mutate()
ui.stack(…).describe()     // → one screen of YAML; what the agent reads

const taggable = fragment('taggable')
  .attr('tags', [])                                                // a new attribute on the node
  .method('tag', (b, ...tags) => b.set('tags', t => [...new Set([...t, ...tags])]))   // build-time method
  .method('untag', (b, tag) => b.set('tags', t => t.filter(x => x !== tag)))
  .query('hasTag', (node, tag) => node.attrs.tags.includes(tag))   // readable from IR and from the agent
  .decorate(b => b.strip(ui.row(ui.each(c => c.attr('tags'), t => ui.chip(t).on('click', (_, c) => c.board.filter({ tag: t })))).gap(4)))

const sourced = fragment('sourced')
  .attr('source', null)
  .on('paste', (b, data) => b.set('source', { from: data.origin, at: Date.now() }))   // lifecycle hook, build-time
  .decorate(b => b.footer(ui.text(c => `pasted ${ago(c.attr('source').at)} · ${c.attr('source').from}`).small()))

const annotated = fragment('annotated')
  .attr('note', '')
  .decorate(b => b.after(ui.input('note').placeholder('your note on the clip').multiline()))

// Every clip gets all three
board.clipDefaults(c => c.use(taggable).use(sourced).use(annotated))

// Fragments are typed into the builder, so .tag exists only after .use(taggable):

(function(){
  const esc = s => s.replace(/&/g,'&').replace(/</g,'<').replace(/>/g,'>');
  const re = /(\/\/[^\n]*)|(`(?:\\.|[^`\\])*`|'(?:\\.|[^'\\\n])*'|"(?:\\.|[^"\\\n])*")|(=>)|(\b(?:const|let|var|return|if|else|await|async|import|from|export|new|function|true|false|null|undefined)\b)|(\b\d[\d_.]*\b)|([{}()\[\].,;:?!=<>+\-*\/&|%]+)/g;
  function hl(src){
    let out='', last=0, m;
    while((m=re.exec(src))){
      out+=esc(src.slice(last,m.index));
      const [all,c,s,a,k,n,p]=m;
      if(c) out+=`<span class="com">${esc(c)}</span>`;
      else if(s) out+=`<span class="str">${esc(s)}</span>`;
      else if(a) out+=`<span class="arrow">=></span>`;
      else if(k) out+=`<b>${k}</b>`;
      else if(n) out+=`<span class="num">${n}</span>`;
      else out+=`<span class="punc">${esc(p)}</span>`;
      last=m.index+all.length;
    }
    return out+esc(src.slice(last));
  }
  document.querySelectorAll('script[type="text/x-src"]').forEach(s=>{
    const pre=document.createElement('pre'); pre.className='src';
    pre.innerHTML=hl(s.textContent.replace(/^\n/,''));
    s.replaceWith(pre);
  });
})();
